import cv2

cap = cv2.VideoCapture(0)

# 测试不同帧率
test_fps = [10, 15, 30, 60, 90, 120]
supported = []

for fps in test_fps:
    cap.set(cv2.CAP_PROP_FPS, fps)
    actual = cap.get(cv2.CAP_PROP_FPS)
    print(f"设置 {fps}fps → 实际 {actual:.2f}fps")
    if actual > 0:
        supported.append(actual)

cap.release()

if supported:
    max_fps = max(supported)
    print(f"\n✅ 摄像头最高支持: {max_fps:.0f}fps")
else:
    print("❌ 无法获取帧率信息")
